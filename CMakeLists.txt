cmake_minimum_required(VERSION 3.25)
project(omarchy_zones VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(OMARCHY_ZONES_BUILD_PLUGIN "Build the native Hyprland plugin" ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(TOMLPLUSPLUS REQUIRED IMPORTED_TARGET tomlplusplus)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
if(OMARCHY_ZONES_BUILD_PLUGIN)
  pkg_check_modules(HYPRLAND REQUIRED IMPORTED_TARGET hyprland)
  add_library(omarchy-zones SHARED src/plugin.cpp)
  target_link_libraries(omarchy-zones PRIVATE PkgConfig::HYPRLAND PkgConfig::TOMLPLUSPLUS)
  target_compile_options(omarchy-zones PRIVATE -Wall -Wextra -Wno-unused-parameter -fno-gnu-unique)
  set_target_properties(omarchy-zones PROPERTIES PREFIX "")
  find_file(HYPRLAND_VERSION_HEADER version.h HINTS ${HYPRLAND_INCLUDE_DIRS}
    PATH_SUFFIXES src NO_DEFAULT_PATH REQUIRED)
  file(STRINGS "${HYPRLAND_VERSION_HEADER}" HYPR_HASH_LINE REGEX "^#define GIT_COMMIT_HASH")
  string(REGEX REPLACE ".*\"([0-9a-f]+)\".*" "\\1" HYPR_HASH "${HYPR_HASH_LINE}")
  string(LENGTH "${HYPR_HASH}" HYPR_HASH_LENGTH)
  if(NOT HYPR_HASH_LENGTH EQUAL 40 OR NOT HYPR_HASH MATCHES "^[0-9a-f]+$")
    message(FATAL_ERROR "The Hyprland headers do not provide a valid commit hash")
  endif()
  file(WRITE "${CMAKE_BINARY_DIR}/hyprland-hash.txt" "${HYPR_HASH}\n")
else()
  # A test-only build must never retain a stale ABI marker for installation.
  file(REMOVE "${CMAKE_BINARY_DIR}/hyprland-hash.txt")
endif()
add_executable(omarchy-zones-editor src/editor.cpp)
target_link_libraries(omarchy-zones-editor PRIVATE Qt6::Widgets Qt6::Network PkgConfig::TOMLPLUSPLUS)
target_compile_options(omarchy-zones-editor PRIVATE -Wall -Wextra)
include(CTest)
if(BUILD_TESTING)
  if(EXISTS "${CMAKE_SOURCE_DIR}/tests/geometry_test.cpp")
    add_executable(geometry-test tests/geometry_test.cpp)
    target_include_directories(geometry-test PRIVATE src)
    target_link_libraries(geometry-test PRIVATE Qt6::Widgets)
    target_compile_options(geometry-test PRIVATE -UNDEBUG)
    add_test(NAME geometry COMMAND geometry-test)
  endif()
  add_executable(profiles-test tests/profiles_test.cpp)
  target_include_directories(profiles-test PRIVATE src)
  target_compile_options(profiles-test PRIVATE -Wall -Wextra -UNDEBUG)
  add_test(NAME profiles COMMAND profiles-test)
  add_executable(editor-profiles-test tests/editor_profiles_test.cpp)
  target_link_libraries(editor-profiles-test PRIVATE Qt6::Widgets Qt6::Network PkgConfig::TOMLPLUSPLUS)
  target_compile_options(editor-profiles-test PRIVATE -Wall -Wextra)
  add_test(NAME editor-profiles COMMAND editor-profiles-test)
  set_tests_properties(editor-profiles PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  add_executable(theme-test tests/theme_test.cpp)
  target_include_directories(theme-test PRIVATE src)
  target_link_libraries(theme-test PRIVATE PkgConfig::TOMLPLUSPLUS)
  target_compile_options(theme-test PRIVATE -Wall -Wextra)
  add_test(NAME theme COMMAND theme-test)
  set_tests_properties(theme PROPERTIES TIMEOUT 10)
  add_executable(editor-native-check tests/editor_native.cpp)
  target_link_libraries(editor-native-check PRIVATE Qt6::Widgets Qt6::Network PkgConfig::TOMLPLUSPLUS)
  target_compile_options(editor-native-check PRIVATE -Wall -Wextra)
  add_executable(editor-instance-test tests/editor_instance_test.cpp)
  target_link_libraries(editor-instance-test PRIVATE Qt6::Core Qt6::Network)
  target_compile_options(editor-instance-test PRIVATE -Wall -Wextra)
  add_test(NAME editor-instance COMMAND editor-instance-test)
  set_tests_properties(editor-instance PROPERTIES TIMEOUT 45)
  add_executable(editor-render-test tests/editor_render_test.cpp)
  target_link_libraries(editor-render-test PRIVATE Qt6::Widgets PkgConfig::TOMLPLUSPLUS)
  target_compile_options(editor-render-test PRIVATE -Wall -Wextra)
  add_test(NAME editor-render COMMAND editor-render-test)
  set_tests_properties(editor-render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 20)
  add_test(NAME editor-render-hidpi COMMAND editor-render-test)
  set_tests_properties(editor-render-hidpi PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_SCALE_FACTOR=2" TIMEOUT 20)
endif()
